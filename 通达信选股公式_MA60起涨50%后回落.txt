{===========================================================}
{ 公式名称：MA60起涨50%后回落                                  }
{ 类型：条件选股公式   适用周期：日线                          }
{ 形态：近3个月（60个交易日）内，股价相对MA60的最大涨幅曾≥50%， }
{       随后回落至MA60上方5%以内（含已跌破MA60的）             }
{ 说明：本文件可整段复制粘贴，中文说明均为注释，不影响编译     }
{===========================================================}

MA60:=MA(C,60);
K:=60;                          {3个月约60个交易日}

A:=(H-MA60)/MA60*100;           {每日最高价相对当日MA60的涨幅%}
T1:=HHV(A,K)>=50;               {曾从MA60起涨超50%}
T2:=C<=MA60*1.05;               {当前回落到MA60上方5%以内，已跌破MA60同样满足}
T3:=NOT(NAMELIKE('ST'));        {剔除ST/*ST}

XG:T1 AND T2 AND T3;


{===========================================================}
{ 备用方案二：先定位最高价，再用当天的MA60作基准（更直观）     }
{ 用法：把下面代码替换上面的选股主体（删掉原 XG 行）           }
{===========================================================}
{ K:=60;                                                      }
{ BARS:=HHVBARS(H,K);        近K日最高价距今的天数            }
{ BASE:=REF(MA60,BARS);      最高价那天的MA60                 }
{ T1:=H>=BASE*1.5;           最高价相对起涨时MA60涨幅≥50%     }
{ T2:=C<=MA60*1.05;                                           }
{ T3:=NOT(NAMELIKE('ST'));                                    }
{ XG:T1 AND T2 AND T3;                                        }

{===========================================================}
{ 备用方案三：加强版，要求起涨前确实贴近过MA60，形态更纯       }
{===========================================================}
{ MA60:=MA(C,60);                                             }
{ K:=60;                                                      }
{ A:=(H-MA60)/MA60*100;                                       }
{ MN:=LLV(A,K);              近K日相对MA60的最小涨幅，负值=曾跌破 }
{ T0:=MN<=10;                起涨前曾贴近MA60（偏离≤10%）     }
{ T1:=HHV(A,K)>=50;                                           }
{ T2:=C<=MA60*1.05;                                           }
{ T3:=NOT(NAMELIKE('ST'));                                    }
{ T4:=BARSCOUNT(C)>120;      上市满120个交易日，排除新股       }
{ XG:T0 AND T1 AND T2 AND T3 AND T4;                          }

{===========================================================}
{ 可选追加条件：按需用 AND 接到 XG 末尾                        }
{===========================================================}
{ AND MA60>REF(MA60,1)        MA60向上，趋势未走坏             }
{ AND V<MA(V,5)               缩量回踩                        }
{ AND C<=HHV(H,K)*0.75        距最高点已回撤25%以上            }
{ AND NOT(NAMELIKE('退'))     排除退市整理股                   }

{===========================================================}

{ 2. 粘贴代码 → 公式名如 MA60_FALLBACK → 确定                  }
{ 3. Ctrl+T 条件选股 → 选该公式 → 周期选日线 → 执行选股        }
