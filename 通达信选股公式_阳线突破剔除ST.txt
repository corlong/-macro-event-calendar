{============================================}
{ 公式名称：阳线突破（剔除ST）              }
{ 类型：条件选股公式  适用周期：日线        }
{ 条件：涨幅2%~9.99% + 昨阴今阳 + 今收>昨开 + 非ST }
{============================================}

{===== 涨幅 2% ~ 9.99% =====}
ZF:=(C-REF(C,1))/REF(C,1)*100;
T1:=ZF>=2 AND ZF<=9.99;

{===== 昨天阴线 且 今天阳线 =====}
T2:=REF(C,1)<REF(O,1) AND C>O;

{===== 今天收盘价 > 昨天开盘价 =====}
T3:=C>REF(O,1);

{===== 剔除ST / *ST =====}
T4:=NOT(NAMELIKE('ST'));

{===== 综合输出 =====}
XG:T1 AND T2 AND T3 AND T4;
